\section{Why absolute nothing cannot exist}

Let $\mathsf N$ denote absolute nothing, defined by absence of any relation, distinction or referent:
\begin{equation}
\mathsf N:=\neg\exists R.
\end{equation}
TIR uses relational existence typing:
\begin{equation}
\operatorname{Exist}(x)\Longrightarrow\exists R_x.
\end{equation}
Hence
\begin{equation}
\operatorname{Exist}(\mathsf N)
\Longrightarrow
\exists R_{\mathsf N}
\end{equation}
contradicts
\begin{equation}
\mathsf N\Longrightarrow\neg\exists R.
\end{equation}
Therefore
\begin{equation}
\boxed{\neg\operatorname{Exist}(\mathsf N).}
\label{eq:v12_nothing_nonrealizable}
\end{equation}

The mathematical value $0$ is not identified with $\mathsf N$. It remains a typed relational nullity, for example
\begin{equation}
\Delta_R(a,a)=0.
\end{equation}

\section{Two distinct minima}

The minimum non-empty object carrier has one element and is represented by a point:
\begin{equation}
\boxed{\min(\text{object})=P.}
\label{eq:v12_point_minimality}
\end{equation}
Equivalently, in ordinary topological typing, the minimum object dimension is zero.

A point alone has no nontrivial internal comparison. The minimum nontrivial structure is therefore a relation:
\begin{equation}
\boxed{\min(\text{nontrivial structure})=R.}
\label{eq:v12_relation_minimality}
\end{equation}

Thus point minimality and relational primacy are not competing claims:
\begin{equation}
\boxed{
\min(\text{object})=P,
\qquad
\min(\text{nontrivial structure})=R.
}
\end{equation}

\section{Primitive relation and binary orbit}

For a non-self-collapsed relation, source and target roles are distinct:
\begin{equation}
s(R)\neq t(R).
\end{equation}
Orientation reversal is the involution
\begin{equation}
J:(s,t;R)\mapsto(t,s;R^{-1}),
\qquad
J^2=\mathrm{id}.
\end{equation}
Rename the two orientation roles
\begin{equation}
\boxed{N=[s\to t],\qquad S=[t\to s].}
\end{equation}
Therefore
\begin{equation}
\boxed{\mathcal O_J=\{N,S\}.}
\label{eq:v12_binary_orbit}
\end{equation}

\begin{theorem}[Primitive binary orbit]
For a nontrivial relational dyad with orientation reversal $J$, the orbit under $\langle J\rangle$ contains exactly the two orientation roles $N$ and $S$.
\end{theorem}
