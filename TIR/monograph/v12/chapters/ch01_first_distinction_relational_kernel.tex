\begin{equation}

\label{eq:v12_zero_axiom_count}

\begin{equation}
\boxed{\mathsf{Nothing}\notin\mathcal O,\qquad 0\neq\mathsf{Nothing}.}

\section{Primitive exchange orbit}

A nontrivial relation has two orientation roles. Define orientation reversal
\begin{equation}
J:(a,b;R)\mapsto(b,a;R^{-1}),
\qquad
J^2=\mathrm{id}.
\end{equation}
For a non-self-collapsed distinction the two orientation classes are distinct. Rename them
\begin{equation}
\boxed{N=[a\to b],\qquad S=[b\to a].}
\label{eq:v12_primitive_binary_orbit}
\end{equation}
The exchange orbit is
\begin{equation}
\boxed{\mathcal O_J=\{N,S\}.}
\end{equation}

\begin{theorem}[Primitive binary orbit]
For a nontrivial relational dyad with orientation reversal $J$, the orbit under $\langle J\rangle$ contains exactly the two orientation roles $N$ and $S$.
\end{theorem}

The pole pair is therefore not inserted as an independent axiom; it is the two-role closure of the primitive relation.
